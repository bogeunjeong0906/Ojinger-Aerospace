set mySlider:ONCHANGE to whenMySliderChanges@.

function whenMySliderChanges {
  parameter newValue.

  print "Value is " +
         round(100*(newValue-mySlider:min)/(mySlider:max-mySlider:min)) +
         "percent of the way between min and max.".
}
